Table users {
  user_id int [pk, increment]
  name varchar(100)
  email varchar(150) [unique, not null]
  password varchar(255) [not null]
  role varchar(20) [not null]
  phone varchar(20)
  profile_image varchar(255)
  status varchar(20)
  created_at datetime
}

Table client_profiles {
  client_id int [pk, increment]
  user_id int [unique, not null]
  company_name varchar(150)
  company_description text
  location varchar(100)
  created_at datetime
}

Table freelancer_profiles {
  freelancer_id int [pk, increment]
  user_id int [unique, not null]
  bio text
  skills text
  experience_years int
  hourly_rate decimal(10,2)
  availability varchar(50)
  portfolio_url varchar(255)
  created_at datetime
}

Table jobs {
  job_id int [pk, increment]
  client_id int [not null]
  title varchar(200)
  description text
  required_skills text
  budget decimal(10,2)
  deadline date
  status varchar(30)
  created_at datetime
}

Table applications {
  application_id int [pk, increment]
  job_id int [not null]
  freelancer_id int [not null]
  proposal text
  proposed_price decimal(10,2)
  status varchar(30)
  applied_at datetime
}

Table projects {
  project_id int [pk, increment]
  job_id int [not null]
  client_id int [not null]
  freelancer_id int [not null]
  project_name varchar(200)
  start_date date
  deadline date
  status varchar(30)
  progress int
  created_at datetime
}

Table milestones {
  milestone_id int [pk, increment]
  project_id int [not null]
  title varchar(200)
  description text
  amount decimal(10,2)
  deadline date
  status varchar(30)
  completed_at datetime
}

Table tasks {
  task_id int [pk, increment]
  milestone_id int [not null]
  title varchar(200)
  description text
  assigned_to int [not null]
  priority varchar(20)
  status varchar(30)
  deadline date
  created_at datetime
}

Table payments {
  payment_id int [pk, increment]
  project_id int [not null]
  milestone_id int
  payer_id int [not null]
  receiver_id int [not null]
  amount decimal(10,2)
  payment_status varchar(30)
  transaction_id varchar(100) [unique]
  payment_date datetime
}

Table ratings_reviews {
  review_id int [pk, increment]
  project_id int [not null]
  reviewer_id int [not null]
  freelancer_id int [not null]
  rating int
  comments text
  created_at datetime
}

Table performance_scores {
  score_id int [pk, increment]
  freelancer_id int [unique, not null]
  rating_score decimal(5,2)
  completion_score decimal(5,2)
  delivery_score decimal(5,2)
  experience_score decimal(5,2)
  feedback_score decimal(5,2)
  overall_score decimal(5,2)
  updated_at datetime
}

Table notifications {
  notification_id int [pk, increment]
  user_id int [not null]
  message text
  type varchar(50)
  is_read boolean
  created_at datetime
}

Table reports {
  report_id int [pk, increment]
  reporter_id int [not null]
  reported_user_id int [not null]
  reason text
  status varchar(30)
  created_at datetime
}

Table ai_matching_results {
  match_id int [pk, increment]
  job_id int [not null]
  freelancer_id int [not null]
  match_score decimal(5,2)
  skill_score decimal(5,2)
  experience_score decimal(5,2)
  rating_score decimal(5,2)
  recommendation_rank int
  created_at datetime
}


// =============================
// RELATIONSHIPS
// =============================

// User Profiles
Ref: users.user_id - client_profiles.user_id
Ref: users.user_id - freelancer_profiles.user_id

// Client and Jobs
Ref: client_profiles.client_id < jobs.client_id

// Jobs and Applications
Ref: jobs.job_id < applications.job_id
Ref: freelancer_profiles.freelancer_id < applications.freelancer_id

// Projects
Ref: jobs.job_id - projects.job_id
Ref: client_profiles.client_id < projects.client_id
Ref: freelancer_profiles.freelancer_id < projects.freelancer_id

// Project and Milestones
Ref: projects.project_id < milestones.project_id

// Milestones and Tasks
Ref: milestones.milestone_id < tasks.milestone_id
Ref: freelancer_profiles.user_id < tasks.assigned_to

// Payments
Ref: projects.project_id < payments.project_id
Ref: milestones.milestone_id < payments.milestone_id
Ref: users.user_id < payments.payer_id
Ref: users.user_id < payments.receiver_id

// Ratings and Reviews
Ref: projects.project_id < ratings_reviews.project_id
Ref: users.user_id < ratings_reviews.reviewer_id
Ref: freelancer_profiles.freelancer_id < ratings_reviews.freelancer_id

// Performance
Ref: freelancer_profiles.freelancer_id - performance_scores.freelancer_id

// Notifications
Ref: users.user_id < notifications.user_id

// Reports
Ref: users.user_id < reports.reporter_id
Ref: users.user_id < reports.reported_user_id

// AI Matching
Ref: jobs.job_id < ai_matching_results.job_id
Ref: freelancer_profiles.freelancer_id < ai_matching_results.freelancer_id
